181:     def simulated_decider(qtype, *args, **kwargs):
182:         choices = {"THINKING_BUDGET": "normal", "UNCERTAINTY_TYPE": "missing_evidence", "NEXT_ACTION": "continue_reasoning"}
183:         return {"status": "simulation", "decision": choices.get(qtype, "insufficient"),
184:                 "probabilities": {}, "confidence": 0.0, "fallback_applied": True}
185:     # Reuse an injected decider, never the caller's research state or counters.
186:     decider = coordinator.governor.decider_fn if coordinator else None
187:     if ablation == "no_jev" or not live_routing:
188:         decider = simulated_decider
189:     with tempfile.TemporaryDirectory(prefix="jev-routing-simulation-") as directory:
190:         coord = ResearchCoordinator(decider_fn=decider, state_directory=Path(directory))
191:         return _simulate_condition(task, ablation, coord, live_routing)

194: def _simulate_condition(task, ablation, coord, live_routing):
195:     start_t = time.perf_counter()
196:     governor = coord.governor
197: 
198:     # 1. Ablation overrides
199:     overrides: dict[str, Any] = {}
200:     if ablation == "no_jev":
201:         # Static heuristic routing
202:         overrides["thinking_budget"] = "normal"
203:         overrides["uncertainty_type"] = "missing_evidence"
204:         overrides["recommended_branch"] = "deepseek_parallel_research"
205: 
206:     plan = coord.plan_task(task["prompt"], overrides=overrides if overrides else None)
207:     budget = plan["thinking_budget"]
208:     route = plan["recommended_branch"]
209: 
210:     # 2. Check stopping heuristic / marginal gains
211:     stopped_by_governor = False
212:     passes_run = 0
213:     total_tokens = 0
214:     total_cost = plan.get("budget_details", {}).get("cost_usd", 0.0)
215: 
216:     if budget == "minimal":
217:         passes_run = 1
218:         total_tokens += 350
219:         total_cost += 0.00002
220:     else:
221:         # Simulate iterative steps evaluated through Cognitive Governor
222:         for p in range(1, 4):
223:             passes_run += 1
224:             total_tokens += 600
225:             total_cost += 0.0001
226:             has_new_evidence = (p == 1) or (task["complexity"] in ("complex", "escalate") and p == 2)
227:             
228:             if ablation == "no_marginal_gain_stop":
229:                 # Stagnation stopping disabled
230:                 next_check = {"decision": "continue_reasoning", "forced_stop_applied": False}
231:             else:
232:                 next_check = governor.check_next_step(
233:                     f"Progress for {task['id']} pass {p}",
234:                     new_evidence_added=has_new_evidence,
235:                     new_hypotheses=1 if has_new_evidence else 0,
236:                 )
237: 
238:             if next_check.get("decision") in ("act", "test") and next_check.get("forced_stop_applied"):
239:                 stopped_by_governor = True
240:                 break
241:             elif next_check.get("decision") in ("act", "test") and not has_new_evidence:
242:                 stopped_by_governor = True
243:                 break
244: 
245:     # Critic check unless ablated
246:     critic_found_issue = False
247:     if ablation != "no_critic" and task["complexity"] in ("complex", "escalate"):
248:         critic_found_issue = True
249: 
250:     # Rework: only occurs if critic was ablated on complex task or wrong route was taken
251:     rework = False
252:     if ablation == "no_critic" and task["complexity"] in ("complex", "escalate"):
253:         rework = True
254:     elif ablation == "no_jev" and task["complexity"] == "escalate":
255:         rework = True
256: 
257:     elapsed = time.perf_counter() - start_t + (0.015 * passes_run)
258:     return {
259:         "measurement_kind": "simulation",
260:         "empirical_performance": False,
261:         "task_id": task["id"],
262:         "mode": f"ablation_{ablation}" if ablation else "controller",
263:         "routing_mode": "live" if live_routing and ablation != "no_jev" else "simulation",
264:         "measured_routing_cost_usd": coord.tracker.total_cost_usd,
265:         "budget": budget,
266:         "route": route,
267:         "passes": passes_run,
268:         "tokens": total_tokens,
269:         "cost_usd": round(total_cost, 6),
270:         "elapsed_seconds": round(elapsed, 4),
271:         "rework": rework,
272:         "critic_found_issue": critic_found_issue,
273:         "stopped_by_governor": stopped_by_governor,
274:         "gating_adjusted": plan.get("gating_adjusted", False),
275:     }

278: def run_benchmark_suite(
279:     tasks: list[dict[str, Any]] | None = None,
280:     include_ablations: bool = True,
281:     quick: bool = False,
282:     live_routing: bool = False,
283: ) -> dict[str, Any]:
284:     """Compare assumed outcomes under simulated routing policies."""
285:     selected_tasks = (tasks or BENCHMARK_TASKS)[: (3 if quick else len(BENCHMARK_TASKS))]
286:     print(f"=== RUNNING SYNTHETIC ROUTING SIMULATION ({len(selected_tasks)} tasks) ===")
287: 
288:     baseline_results = []
289:     controller_results = []
290:     ablations_results: dict[str, list[dict[str, Any]]] = {}
291: 
292:     ablation_types = ["no_jev", "no_critic", "no_marginal_gain_stop"]
293: 
294:     for t in selected_tasks:
295:         print(f"  • Simulating {t['id']}: {t['title'][:45]}...")
296:         # Baseline
297:         b_res = run_single_task_baseline(t)
298:         baseline_results.append(b_res)
299: 
300:         # Controller
301:         c_res = run_single_task_controller(t, live_routing=live_routing)
302:         controller_results.append(c_res)
303: 
304:         # Ablations
305:         if include_ablations:
306:             for abl in ablation_types:
307:                 abl_res = run_single_task_controller(t, ablation=abl, live_routing=live_routing)
308:                 ablations_results.setdefault(abl, []).append(abl_res)
309: 
310:     # Compute aggregate summaries
311:     def summarize(runs: list[dict[str, Any]]) -> dict[str, Any]:
312:         n = len(runs)
313:         if n == 0:
314:             return {}
315:         total_tokens = sum(r["tokens"] for r in runs)
316:         total_cost = sum(r["cost_usd"] for r in runs)
317:         total_elapsed = sum(r["elapsed_seconds"] for r in runs)
318:         rework_count = sum(1 for r in runs if r.get("rework"))
319:         stops_count = sum(1 for r in runs if r.get("stopped_by_governor"))
320:         return {
321:             "task_count": n,
322:             "total_tokens": total_tokens,
323:             "mean_tokens": round(total_tokens / n, 1),
324:             "total_cost_usd": round(total_cost, 6),
325:             "mean_cost_usd": round(total_cost / n, 6),
326:             "mean_latency_sec": round(total_elapsed / n, 4),
327:             "rework_count": rework_count,
328:             "rework_rate_pct": round((rework_count / n) * 100, 1),
329:             "governor_stops_count": stops_count,
330:         }
331: 
332:     summary = {
333:         "measurement_kind": "simulation",
334:         "empirical_performance": False,
335:         "limitations": "Tasks are not executed; costs, tokens and rework are modeled assumptions, not measured improvements.",
336:         "live_routing": live_routing,
337:         "timestamp_utc": datetime.now(timezone.utc).isoformat(),
338:         "task_count": len(selected_tasks),
339:         "baseline": summarize(baseline_results),
340:         "controller": summarize(controller_results),
341:         "ablations": {k: summarize(v) for k, v in ablations_results.items()},
342:         "baseline_runs": baseline_results,
343:         "controller_runs": controller_results,
344:     }
345: 
346:     return summary
